async function openM(slug, src) { const overlay = document.getElementById('overlay'); const epDiv = document.getElementById('eps'); const iframe = document.getElementById('embed-player'); overlay.style.display = 'flex'; epDiv.innerHTML = '

Đang bóc tách dữ liệu từ ' + src.toUpperCase() + '...

'; iframe.src = ""; try { const res = await fetch(`/api/movies?src=${src}&path=${slug}`).then(r => r.json()); let movie, episodes; // PHÂN LOẠI LOGIC CHO TỪNG NGUỒN if (src === 'nguonc') { movie = res.movie; episodes = res.movie.episodes; } else if (src === 'kkphim') { // KKPhim đôi khi trả về cấu trúc giống Ophim nhưng nằm trong data.item movie = res.movie || res.data?.item; episodes = (res.movie ? res.movie.episodes : res.data?.item?.episodes) || []; } else { // Ophim chuẩn movie = res.data.item; episodes = res.data.item.episodes; } document.getElementById('m-name').innerText = movie.name; // Tìm server có dữ liệu link embed let list = []; if (episodes && episodes.length > 0) { const server = episodes.find(sv => (sv.server_data || sv.items || []).length > 0) || episodes[0]; list = server.server_data || server.items || []; } if (list.length === 0) throw new Error("No link"); epDiv.innerHTML = list.map((e, i) => `
Tập ${e.name}
`).join(''); play(list[0].link_embed, epDiv.querySelector('.ep-btn')); } catch(e) { console.error(e); epDiv.innerHTML = `

Lỗi: Nguồn ${src} chưa cập nhật link cho phim này!

`; } }
